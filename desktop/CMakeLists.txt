cmake_minimum_required(VERSION 3.21)
project(OmaPress VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
qt_standard_project_setup()
qt_add_executable(omapress-desktop src/main.cpp src/Bridge.cpp src/Bridge.h)
qt_add_resources(omapress-desktop "qml" PREFIX "/" FILES qml/Main.qml resources/omapress.svg)
target_link_libraries(omapress-desktop PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
target_compile_options(omapress-desktop PRIVATE -Wall -Wextra -Wpedantic)
include(GNUInstallDirs)
install(TARGETS omapress-desktop RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES resources/omapress.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES resources/omapress.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
